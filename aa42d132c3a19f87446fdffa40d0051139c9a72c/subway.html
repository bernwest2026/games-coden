SUBWAY RUNNER

Renne so weit du kannst!
Weiche Hindernissen aus und sammle Münzen.

⬅️ ➡️ Spur wechseln
⬆️ Leertaste = Springen
⬇️ Ducken

Ein kleines Endless-Runner-Spiel

function update(dt) { elapsed += dt; score += dt * 10; speed = Math.min(0.32 + elapsed * 0.003, 0.70); roadOffset = (roadOffset + dt * speed * 0.9) % 1; // Spurwechsel flüssig und zuverlässig player.xLane += (player.lane - player.xLane) * Math.min(1, dt * 12); if (Math.abs(player.lane - player.xLane) < 0.015) { player.xLane = player.lane; } // Springen if (player.jumpV !== 0 || player.jump > 0) { player.jump += player.jumpV * dt; player.jumpV -= 6.5 * dt; if (player.jump <= 0) { player.jump = 0; player.jumpV = 0; } } // Ducken player.slide = Math.max(0, player.slide - dt); // Hindernisse erzeugen spawnTimer -= dt; if (spawnTimer <= 0) { spawnObstacle(); spawnTimer = Math.max(0.85, 1.35 - elapsed * 0.003) + Math.random() * 0.4; } // Münzen erzeugen coinTimer -= dt; if (coinTimer <= 0) { spawnCoinLine(); coinTimer = 1.7 + Math.random() * 1.3; } // Objekte bewegen und Kollisionen überprüfen for (const obj of objects) { const previousDepth = obj.depth; obj.depth += dt * speed; // Nur prüfen, wenn das Objekt den Spielerbereich erreicht. // Der Bereich ist enger, damit du nicht zu früh stirbst. const reachedPlayer = previousDepth <= 0.985 && obj.depth >= 0.985; const sameLane = Math.abs(player.xLane - obj.lane) < 0.24; if ( obj.kind === "coin" && !obj.collected && obj.depth >= 0.94 && obj.depth <= 1.06 && sameLane ) { obj.collected = true; coinCount++; score += 25; } if ( obj.kind === "obstacle" && !obj.checked && reachedPlayer && sameLane ) { obj.checked = true; // Niedrige Hindernisse: darüber springen const safelyJumping = player.jump > 0.28; // Hohe Hindernisse: darunter durchrutschen const safelySliding = player.slide > 0.05; if (obj.type === "low" && !safelyJumping) { gameOver(); return; } if (obj.type === "high" && !safelySliding) { gameOver(); return; } } } // Alte Objekte entfernen objects = objects.filter(obj => obj.depth < 1.15 && !(obj.kind === "coin" && obj.collected) ); scoreEl.textContent = Math.floor(score); coinsEl.textContent = coinCount; }
